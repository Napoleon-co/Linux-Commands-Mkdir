    pwd — Print working directory path.
    ls — List directory contents.
    cd — Change the current directory.
    mkdir — Create a new directory.
    rmdir — Remove an empty directory.
    touch — Create an empty file.
    cp — Copy files or directories.
    mv — Move or rename files.
    rm — Delete files or directories.
    ln — Create symbolic or hard links.

    cat — Display file contents entirely.
    less — View file contents interactively.
    head — Output the first few lines.
    tail — Output the last few lines.
    grep — Search for specific text patterns.
    awk — Scan and process text patterns.
    sed — Stream editor to modify text.
    sort — Sort lines of text files.
    uniq — Omit or report repeated lines.
    wc — Count lines, words, and characters.
    diff — Compare two files line-by-line.
    nano — Open a simple text editor.
